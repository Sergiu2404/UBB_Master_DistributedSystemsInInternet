
# Connecting to MySQL using Dropwizard

This document provides a step-by-step guide to connect a Dropwizard application to a MySQL database using Hibernate.

---

## **1. Add Dependencies**

Add the following dependencies to your `pom.xml`:

```xml
<dependency>
    <groupId>io.dropwizard</groupId>
    <artifactId>dropwizard-db</artifactId>
    <version>4.0.11</version> <!-- Replace with your Dropwizard version -->
</dependency>
<dependency>
    <groupId>mysql</groupId>
    <artifactId>mysql-connector-java</artifactId>
    <version>9.1.0</version> <!-- Replace with the desired version -->
</dependency>
```

---

## **2. Configuration Class**

Extend your configuration class to include database settings using Dropwizard's `DataSourceFactory`:

```java
package com.example;

import com.fasterxml.jackson.annotation.JsonProperty;
import io.dropwizard.Configuration;
import io.dropwizard.db.DataSourceFactory;

import javax.validation.Valid;
import javax.validation.constraints.NotNull;

public class MyConfiguration extends Configuration {
    @Valid
    @NotNull
    private DataSourceFactory database = new DataSourceFactory();

    @JsonProperty("database")
    public DataSourceFactory getDatabase() {
        return database;
    }

    @JsonProperty("database")
    public void setDatabase(DataSourceFactory database) {
        this.database = database;
    }
}
```

---

## **3. Add Database Configuration in YAML**

Add the database configuration in `config.yml`:

```yaml
database:
  driverClass: com.mysql.cj.jdbc.Driver
  user: your-username
  password: your-password
  url: jdbc:mysql://localhost:3306/your_database_name
  properties:
    charSet: UTF-8
  maxWaitForConnection: 1s
  validationQuery: "SELECT 1"
  minSize: 5
  maxSize: 20
```

Replace `your-username`, `your-password`, and `your_database_name` with your actual database details.

---

## **4. Create an Entity Class**

Define a class to represent your database table. For example:

```java
package com.example.models;

import javax.persistence.*;

@Entity
@Table(name = "users")
public class User {
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Column(name = "name", nullable = false)
    private String name;

    @Column(name = "email", nullable = false, unique = true)
    private String email;

    // Getters and Setters
    public Long getId() {
        return id;
    }

    public void setId(Long id) {
        this.id = id;
    }

    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }

    public String getEmail() {
        return email;
    }

    public void setEmail(String email) {
        this.email = email;
    }
}
```

---

## **5. DAO Class**

Create a DAO (Data Access Object) class using Hibernate:

```java
package com.example.dao;

import com.example.models.User;
import io.dropwizard.hibernate.AbstractDAO;
import org.hibernate.SessionFactory;

import java.util.List;

public class UserDAO extends AbstractDAO<User> {
    public UserDAO(SessionFactory sessionFactory) {
        super(sessionFactory);
    }

    public List<User> findAll() {
        return list(criteria());
    }

    public User findById(Long id) {
        return get(id);
    }

    public User create(User user) {
        return persist(user);
    }
}
```

---

## **6. Register Hibernate in the Application Class**

Update your `MyApplication` class:

```java
package com.example;

import com.example.dao.UserDAO;
import com.example.models.User;
import io.dropwizard.Application;
import io.dropwizard.hibernate.HibernateBundle;
import io.dropwizard.setup.Bootstrap;
import io.dropwizard.setup.Environment;

public class MyApplication extends Application<MyConfiguration> {
    private final HibernateBundle<MyConfiguration> hibernateBundle =
            new HibernateBundle<MyConfiguration>(User.class) {
                @Override
                public io.dropwizard.db.DataSourceFactory getDataSourceFactory(MyConfiguration configuration) {
                    return configuration.getDatabase();
                }
            };

    public static void main(String[] args) throws Exception {
        new MyApplication().run(args);
    }

    @Override
    public void initialize(Bootstrap<MyConfiguration> bootstrap) {
        bootstrap.addBundle(hibernateBundle);
    }

    @Override
    public void run(MyConfiguration configuration, Environment environment) {
        final UserDAO userDAO = new UserDAO(hibernateBundle.getSessionFactory());

        // Register Resources
        environment.jersey().register(new UserResource(userDAO));
    }
}
```

---

## **7. Resource Class**

Create a resource class to expose the database operations via REST:

```java
package com.example.resources;

import com.example.dao.UserDAO;
import com.example.models.User;

import javax.ws.rs.*;
import javax.ws.rs.core.MediaType;
import java.util.List;

@Path("/users")
@Produces(MediaType.APPLICATION_JSON)
@Consumes(MediaType.APPLICATION_JSON)
public class UserResource {
    private final UserDAO userDAO;

    public UserResource(UserDAO userDAO) {
        this.userDAO = userDAO;
    }

    @GET
    @UnitOfWork
    public List<User> getUsers() {
        return userDAO.findAll();
    }

    @POST
    @UnitOfWork
    public User createUser(User user) {
        return userDAO.create(user);
    }
}
```

The `@UnitOfWork` annotation in Dropwizard is a powerful tool for managing Hibernate sessions within the context of REST resource methods.

---

## **8. Running the Application**

1. Compile the application:
   ```bash
   mvn clean package
   ```
2. Run the application:
   ```bash
   java -jar target/your-app-name-1.0-SNAPSHOT.jar server config.yml
   ```
3. Access the REST endpoint:
   - **Get Users**: `http://localhost:8080/users`
   - **Create User**: Use POST on the same URL with JSON payload like:
     ```json
     {
         "name": "John Doe",
         "email": "john.doe@example.com"
     }
     ```
